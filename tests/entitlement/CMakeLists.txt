cmake_minimum_required(VERSION 3.20)
project(EntitlementChecks LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(PORT_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
add_executable(entitlement_fixture bridge.cpp "${PORT_ROOT}/entitlement.cpp" "${PORT_ROOT}/entitlement_edat.c")
target_include_directories(entitlement_fixture PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PORT_ROOT}" "${PORT_ROOT}/external/ps3recomp/libs/filesystem")
if(MSVC)
    target_compile_definitions(entitlement_fixture PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
enable_testing()
foreach(suite IN ITEMS test_entitlement test_prepare_entitlement)
    add_test(NAME ${suite} COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${PORT_ROOT}/scripts" -p "${suite}.py")
    set_tests_properties(${suite} PROPERTIES ENVIRONMENT "TB_ENTITLEMENT_TEST_RUNNER=$<TARGET_FILE:entitlement_fixture>" TIMEOUT 90)
endforeach()
