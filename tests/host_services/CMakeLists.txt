cmake_minimum_required(VERSION 3.20)
project(HostServicesChecks LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(PORT_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(SDK "${PORT_ROOT}/external/ps3recomp")
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
find_package(SDL2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
set(OVERLAY "${CMAKE_CURRENT_BINARY_DIR}/generated/overlay")
add_executable(hle_fixture hle.cpp "${OVERLAY}/runtime/ppu/ppu_hle.cpp" "${PORT_ROOT}/stubs.cpp" "${PORT_ROOT}/hle_inventory.cpp" "${PORT_ROOT}/execution_policy.cpp" "${PORT_ROOT}/boot_diagnostics.cpp")
target_include_directories(hle_fixture PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${PORT_ROOT}" "${SDK}/include")
if(MSVC)
    target_compile_options(hle_fixture PRIVATE /UNDEBUG)
    target_compile_definitions(hle_fixture PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
    target_compile_options(hle_fixture PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME hle_toc_boundary COMMAND hle_fixture)
foreach(name IN ITEMS positions hotplug keyboard audio cache)
    add_executable(${name}_fixture "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}.c")
    target_include_directories(${name}_fixture PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${PORT_ROOT}" "${SDK}/include")
    target_link_libraries(${name}_fixture PRIVATE SDL2::SDL2 Threads::Threads)
    target_compile_definitions(${name}_fixture PRIVATE SDL_MAIN_HANDLED NOMINMAX)
    if(NOT name STREQUAL "cache")
        target_sources(${name}_fixture PRIVATE "${PORT_ROOT}/input_events.cpp")
    endif()
    if(MSVC)
        target_compile_options(${name}_fixture PRIVATE /UNDEBUG)
        target_compile_definitions(${name}_fixture PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_RAND_S)
        if(name STREQUAL "audio")
            target_link_libraries(${name}_fixture PRIVATE winmm)
        endif()
    else()
        target_compile_options(${name}_fixture PRIVATE -UNDEBUG)
        target_link_libraries(${name}_fixture PRIVATE m)
    endif()
    if(NOT name STREQUAL "cache")
        add_test(NAME ${name} COMMAND ${name}_fixture)
        set_tests_properties(${name} PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy;SDL_JOYSTICK_HIDAPI=0;SDL_GAMECONTROLLER_IGNORE_DEVICES_EXCEPT=0x0000/0x0000" TIMEOUT 30)
    endif()
endforeach()
add_test(NAME audio_no_device COMMAND audio_fixture no-device)
set_tests_properties(audio_no_device PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=nonexistent-test-driver" TIMEOUT 30)
add_test(NAME cache_publication COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${PORT_ROOT}/scripts" -p test_edat_cache.py)
set_tests_properties(cache_publication PROPERTIES ENVIRONMENT "TB_CACHE_TEST_RUNNER=$<TARGET_FILE:cache_fixture>" TIMEOUT 60)
add_test(NAME hle_strict COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify.py" "$<TARGET_FILE:hle_fixture>")
add_executable(timeline_fixture timeline.cpp "${PORT_ROOT}/frame_budget.cpp")
target_include_directories(timeline_fixture PRIVATE "${PORT_ROOT}" "${SDK}/include")
target_link_libraries(timeline_fixture PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(timeline_fixture PRIVATE /UNDEBUG)
    target_compile_definitions(timeline_fixture PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(timeline_fixture PRIVATE -UNDEBUG)
endif()
add_test(NAME frame_budget_trigger COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_timeline.py" "$<TARGET_FILE:timeline_fixture>")
add_test(NAME frame_budget_analysis COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${PORT_ROOT}/scripts" -p test_frame_budget.py)
add_subdirectory("${PORT_ROOT}/tests/entitlement" entitlement)
