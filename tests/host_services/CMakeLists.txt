foreach(name IN ITEMS positions hotplug keyboard audio cache)
    add_executable(${name}_fixture "${CMAKE_CURRENT_BINARY_DIR}/generated/${name}.c")
    target_include_directories(${name}_fixture PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${PORT_ROOT}" "${SDK}/include")
    target_link_libraries(${name}_fixture PRIVATE SDL2::SDL2 Threads::Threads)
    target_compile_definitions(${name}_fixture PRIVATE SDL_MAIN_HANDLED NOMINMAX)
    if(NOT name STREQUAL "cache")
        target_sources(${name}_fixture PRIVATE "${PORT_ROOT}/input_events.cpp")
    endif()
    if(MSVC)
        target_compile_options(${name}_fixture PRIVATE /UNDEBUG)
        target_compile_definitions(${name}_fixture PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_RAND_S)
        if(name STREQUAL "audio")
            target_link_libraries(${name}_fixture PRIVATE winmm)
        endif()
    else()
        target_compile_options(${name}_fixture PRIVATE -UNDEBUG)
        target_link_libraries(${name}_fixture PRIVATE m)
    endif()
    if(NOT name STREQUAL "cache")
        add_test(NAME ${name} COMMAND ${name}_fixture)
        set_tests_properties(${name} PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy;SDL_JOYSTICK_HIDAPI=0;SDL_GAMECONTROLLER_IGNORE_DEVICES_EXCEPT=0x0000/0x0000" TIMEOUT 30)
    endif()
endforeach()

# Compile the shared VMX helpers with the same semantics and baseline ISA as
# the lifted title, including on the Windows clang-cl runner.

    add_executable(vmx_${mode}_fixture "${SDK}/tests/test_vmx_fmaf.c")
    tb_title_compile_flags(vmx_${mode}_fixture)
    if(mode STREQUAL "scalar")
        target_compile_definitions(vmx_${mode}_fixture PRIVATE VMX_SCALAR_FMA)
    endif()
    if(NOT WIN32)
        target_link_libraries(vmx_${mode}_fixture PRIVATE m)
    endif()
    add_test(NAME vmx_${mode}_fused COMMAND vmx_${mode}_fixture)
    set_tests_properties(vmx_${mode}_fused PROPERTIES TIMEOUT 120)
endforeach()
