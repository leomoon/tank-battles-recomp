cmake_minimum_required(VERSION 3.20)
project(RpcnRoomChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(PORT_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(SDK "${PORT_ROOT}/external/ps3recomp")
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
find_package(CURL 7.77 REQUIRED)
find_package(Threads REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${PORT_ROOT}/scripts/runtime_abi.py"
    --sdk "${SDK}" --output "${CMAKE_CURRENT_BINARY_DIR}/abi" COMMAND_ERROR_IS_FATAL ANY)
add_executable(matching_rooms rooms.cpp "${PORT_ROOT}/rpcn/client.cpp"
    "${PORT_ROOT}/rpcn/matching.cpp" "${PORT_ROOT}/rpcn/matching_room.cpp")
target_include_directories(matching_rooms PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/abi" "${PORT_ROOT}" "${SDK}/include")
target_compile_definitions(matching_rooms PRIVATE PPU_NO_INLINE_LOADS NOMINMAX)
target_link_libraries(matching_rooms PRIVATE CURL::libcurl Threads::Threads)
if(MSVC)
    target_compile_options(matching_rooms PRIVATE /UNDEBUG)
else()
    target_compile_options(matching_rooms PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME matching_wire COMMAND matching_rooms --wire)
