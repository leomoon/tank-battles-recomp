cmake_minimum_required(VERSION 3.20)
project(TextureDecodeChecks LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(PORT_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(SDK "${PORT_ROOT}/external/ps3recomp" CACHE PATH "Vendored ps3recomp source")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${PORT_ROOT}/scripts/instrument_sdk.py" --sdk "${SDK}" --output "${CMAKE_CURRENT_BINARY_DIR}/overlay" COMMAND_ERROR_IS_FATAL ANY)
enable_testing()
foreach(mode IN ITEMS specialized scalar)
 add_executable(decode_${mode} "${PORT_ROOT}/tests/texture_decode.cpp" "${PORT_ROOT}/texture_decode.cpp" "${PORT_ROOT}/texture_decode_fast.cpp" "${CMAKE_CURRENT_BINARY_DIR}/overlay/libs/video/rsx_texture_layout.c")
 target_include_directories(decode_${mode} PRIVATE "${PORT_ROOT}" "${SDK}/include" "${SDK}/libs/video")
 if(MSVC)
  target_compile_definitions(decode_${mode} PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(decode_${mode} PRIVATE /UNDEBUG /fp:precise)
 else()
  target_compile_options(decode_${mode} PRIVATE -UNDEBUG -ffp-contract=off)
 endif()
 if(mode STREQUAL "scalar")
  target_compile_definitions(decode_${mode} PRIVATE TB_TEXTURE_DECODE_NO_SIMD=1)
 endif()
 add_test(NAME decode_${mode} COMMAND decode_${mode})
 add_test(NAME reference_${mode} COMMAND decode_${mode})
 set_tests_properties(reference_${mode} PROPERTIES ENVIRONMENT "TB_TEXTURE_DECODE_MODE=reference")
endforeach()
# Preserve the upstream expected-value layout/format contract as an independent
# complement to the old-versus-new differential oracle.
add_executable(decode_contract "${SDK}/libs/video/tests/test_texture_layout.c" "${PORT_ROOT}/texture_decode.cpp" "${PORT_ROOT}/texture_decode_fast.cpp" "${CMAKE_CURRENT_BINARY_DIR}/overlay/libs/video/rsx_texture_layout.c")
target_include_directories(decode_contract PRIVATE "${PORT_ROOT}" "${SDK}/include" "${SDK}/libs/video")
if(MSVC)
 target_compile_definitions(decode_contract PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(decode_contract PRIVATE /UNDEBUG /fp:precise)
else()
 target_compile_options(decode_contract PRIVATE -UNDEBUG -ffp-contract=off)
endif()
find_package(Threads REQUIRED)
foreach(mode IN ITEMS specialized scalar contract)
 target_link_libraries(decode_${mode} PRIVATE Threads::Threads)
endforeach()
add_test(NAME expected_layout_contract COMMAND decode_contract)
